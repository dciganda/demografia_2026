pop[,i+2]  <- Mf %*% pop[,i]                        # mujeres en t+5; el 1.er elemento es B_totpop[,i+3]  <- Mm %*% pop[,i+1]                      # varones en t+5, supervivencia masculinabirths     <- pop[1,i+2]                            # guardar los nacimientos totalespop[1,i+2] <- births / (1 + srb) * Lf[1]/int        # niñas de 0–4pop[1,i+3] <- births * srb / (1 + srb) * Lm[1]/int  # varones de 0–4
